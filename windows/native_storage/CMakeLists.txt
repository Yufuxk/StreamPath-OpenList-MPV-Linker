cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(streampath_storage LANGUAGES C CXX)
else()
  enable_language(C)
endif()
include(FetchContent)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_UTILS OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBKRB5 OFF CACHE BOOL "" FORCE)
set(ENABLE_GSSAPI OFF CACHE BOOL "" FORCE)
set(CORE_LIBRARIES "" CACHE INTERNAL "" FORCE)
FetchContent_Declare(sp_smb2 URL https://codeload.github.com/sahlberg/libsmb2/zip/refs/tags/libsmb2-6.2
  URL_HASH SHA256=51668eac2b9cde94869a8f45f4bb240b265ca5469db4dfbd9b6a79341b0cc08b
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE TIMEOUT 120 INACTIVITY_TIMEOUT 30 EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(sp_smb2)
FetchContent_Declare(sp_nfs URL https://codeload.github.com/sahlberg/libnfs/zip/refs/tags/libnfs-6.0.2
  URL_HASH SHA256=92a78d63b0cc4e97154dcf2a39bca62569a91350c56e257381fbd0c81c85b64d
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE TIMEOUT 120 INACTIVITY_TIMEOUT 30 EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(sp_nfs)
include(${CMAKE_CURRENT_SOURCE_DIR}/libnfs_patches.cmake)
set_target_properties(nfs PROPERTIES OUTPUT_NAME libnfs)
if(MSVC)
  target_compile_options(nfs PRIVATE /utf-8 /FI${CMAKE_CURRENT_SOURCE_DIR}/nfs_msvc_compat.h)
endif()
target_include_directories(nfs PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
set(USE_LIBIDN2 OFF CACHE BOOL "" FORCE)
set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
set(CURL_USE_LIBSSH OFF CACHE BOOL "" FORCE)
set(CURL_USE_GSSAPI OFF CACHE BOOL "" FORCE)
set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
set(CURL_DISABLE_LDAP ON CACHE BOOL "" FORCE)
set(CURL_DISABLE_LDAPS ON CACHE BOOL "" FORCE)
FetchContent_Declare(sp_curl URL https://curl.se/download/curl-8.22.0.zip
  URL_HASH SHA256=f9ec970e52124e494606209e6bc0e985c623b428796641742f60c5b56931aaee
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE TIMEOUT 120 INACTIVITY_TIMEOUT 30 EXCLUDE_FROM_ALL)
FetchContent_MakeAvailable(sp_curl)
add_library(streampath_storage SHARED storage.cpp)
target_compile_features(streampath_storage PRIVATE cxx_std_17)
target_compile_definitions(streampath_storage PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(streampath_storage PRIVATE /W4 /WX /EHsc /utf-8)
target_include_directories(streampath_storage PRIVATE ${sp_nfs_SOURCE_DIR}/include)
target_link_libraries(streampath_storage PRIVATE smb2 nfs CURL::libcurl)
add_custom_command(TARGET streampath_storage POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:smb2> $<TARGET_FILE_DIR:streampath_storage>
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:nfs> $<TARGET_FILE_DIR:streampath_storage>
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:libcurl_shared> $<TARGET_FILE_DIR:streampath_storage>)
install(TARGETS streampath_storage smb2 nfs libcurl_shared RUNTIME DESTINATION "${CMAKE_INSTALL_PREFIX}" COMPONENT Runtime)
install(FILES ${sp_smb2_SOURCE_DIR}/COPYING DESTINATION "${CMAKE_INSTALL_PREFIX}/licenses/libsmb2" COMPONENT Runtime)
install(FILES ${sp_nfs_SOURCE_DIR}/LICENCE-LGPL-2.1.txt ${sp_nfs_SOURCE_DIR}/LICENCE-BSD.txt DESTINATION "${CMAKE_INSTALL_PREFIX}/licenses/libnfs" COMPONENT Runtime)
install(FILES ${sp_curl_SOURCE_DIR}/COPYING DESTINATION "${CMAKE_INSTALL_PREFIX}/licenses/libcurl" COMPONENT Runtime)
